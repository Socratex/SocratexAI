## New content of `~/.claude/hooks/socratex-pipeline-loader.sh`
```bash
#!/usr/bin/env bash
# SocratexAI session-start loader.
# Fires only when cwd is inside /home/socratex/PhpstormProjects/.
# Prints the ordered read list from `sox context --project <repo root>` (project knowledge lives in SocratexAI/local).
# 2026-11-23: no more pointers to WORK/<PROJECT>; the list comes from the resolver.
set -euo pipefail

cwd="${CLAUDE_PROJECT_DIR:-$PWD}"
root=/home/socratex/PhpstormProjects
sox_home=/home/socratex/PhpstormProjects/SocratexAI
case "$cwd" in
  "$root"|"$root"/*) ;;
  *) exit 0 ;;
esac

# Repo root: the git top level of cwd, else cwd itself.
repo="$(git -C "$cwd" rev-parse --show-toplevel 2>/dev/null || printf '%s' "$cwd")"

SOCRATEX_REPO="$repo" SOX_HOME="$sox_home" WORKSPACE="$root" python3 -I - <<'PY'
import json, os, subprocess
repo, sox_home, root = os.environ["SOCRATEX_REPO"], os.environ["SOX_HOME"], os.environ["WORKSPACE"]
try:
    done = subprocess.run(["python3", f"{sox_home}/core/sox", "context", "--project", repo],
                          capture_output=True, text=True, timeout=20)
    listing = (done.stdout or "").strip() or f"(sox context printed nothing: {(done.stderr or '').strip()[:300]})"
except (OSError, subprocess.SubprocessError) as exc:
    listing = f"(sox context failed: {exc}; run `python3 {sox_home}/core/sox context --project {repo}` by hand)"
lines = [
    "SocratexAI workspace detected. Read these files in this order (output of `sox context --project " + repo + "`; a later row wins on conflict; rows marked conditional only when their condition holds):",
    listing,
    f"Workspace pointer: {root}/CLAUDE.md. Ticket memory: {sox_home}/local/tickets/<PREFIX>/<TICKET>/ (ticket = first [A-Z]{{2,}}-\\d+ in the branch name).",
    "Git: commit own work per the core rules; the owner pushes, never push.",
    "After loading, give one short line confirming it is active.",
]
print(json.dumps({"hookSpecificOutput": {"hookEventName": "SessionStart", "additionalContext": "\n".join(lines)}}))
PY
```
